AFC03 AIC8800 offline installation commands
============================================

# Execute one command at a time. Do not copy comment lines.
# Package directory on the board:

cd "$HOME/afc03-aic8800"

sha256sum -c SHA256SUMS.txt

uname -rm

sudo bash ./install_aic8800_wifi.sh --check ./wifi

# STOP unless the previous command prints CHECK PASSED.

sudo bash ./install_aic8800_wifi.sh --install ./wifi

lsmod | grep -E 'aic8800_fdrv|aic_load_fw'

modprobe --show-depends aic8800_fdrv

dmesg | tail -n 150

WIFI_IF="$(for p in /sys/class/net/*; do [ -d "$p/wireless" ] && basename "$p"; done | head -n 1)"

printf 'Wi-Fi interface: %s\n' "$WIFI_IF"

ip -brief link show dev "$WIFI_IF"

sudo nmtui

ip -brief address show dev "$WIFI_IF"

ip route

ping -c 4 223.5.5.5

ping -c 4 www.baidu.com

# Run persistence only after live Wi-Fi is stable.

sudo bash ./install_aic8800_wifi.sh --persist ./wifi

cat /etc/modules-load.d/aic8800.conf

# Reboot once and repeat the interface/IP/ping checks.


# 2026-09-12: replacement-board checks. Run each stage interactively.
# Full explanation: docs/4gb-replacement-setup.md
# Vim only: press Esc, type :set mouse=, press Enter; drag-select with
# the left mouse button, then middle-click in another board terminal.

# Confirm 4 GB RAM and check the clock; these do not install anything.
free -m

date

timedatectl status

# After Wi-Fi connects:
nmcli device status

hostname -I

getent hosts www.baidu.com

# If the BSP already has a supported time-sync service:
sudo timedatectl set-ntp true

# In nmtui, edit the connected profile and enable automatic connection.
# Module persistence alone does not verify cold-boot reconnection.

# Enable the existing SSH service on the BOARD:
command -v sshd

sudo systemctl enable --now ssh

systemctl is-active ssh

# Read this fingerprint on the board and compare it on the PC:
ssh-keygen -lf /etc/ssh/ssh_host_ed25519_key.pub

# On WINDOWS, replace <BOARD_IP> before running:
# ssh -o HostKeyAlgorithms=ssh-ed25519 linaro@<BOARD_IP>
# Do not disable host-key checking when replacing a board.

# Change the CURRENT BOARD USER password interactively.
# Enter the current password, then the new password twice; no echo is normal.
passwd
